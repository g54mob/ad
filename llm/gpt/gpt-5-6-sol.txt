def split_frontmatter(text: str) -> tuple[str, str]:
    text = text.removeprefix("\ufeff")
    lines = text.splitlines(keepends=True)
    if not lines or lines[0].strip() != "---":
        return "", text
    for index, line in enumerate(lines[1:], start=1):
        if line.strip() == "---":
            return "".join(lines[1:index]), "".join(lines[index + 1:]).lstrip("\r\n")
    raise ValueError("Unclosed YAML frontmatter")


def prompt_description(text: str) -> str:
    frontmatter, body = split_frontmatter(text)
    description = frontmatter_value(frontmatter, "description")
    if description:
        return description
    for line in body.splitlines():
        stripped = line.strip().lstrip("#").strip()
        if stripped:
            return stripped[:160]
    return "(no description)"


class Skill:
    def __init__(self, name: str, description: str, rel_path: str, text: str, aliases: set[str]):
        self.name = name
        self.description = description
        self.rel_path = rel_path
        self.text = text
        self.aliases = aliases


def frontmatter_value(frontmatter: str, key: str) -> str:
    metadata = yaml.safe_load(frontmatter) or {}
    if not isinstance(metadata, dict):
        raise ValueError("Frontmatter must be a YAML mapping")
    value = metadata.get(key, "")
    return ",".join(str(item) for item in value) if isinstance(value, list) else str(value)


def config_roots() -> list[Path]:
    roots: list[Path] = []
    for root in (WORKSPACE, AGENT_HOME):
        resolved = root.resolve()
        if resolved not in roots:
            roots.append(resolved)
    return roots


def instruction_documents(path: Path, seen: set[Path] | None = None) -> list[tuple[Path, str]]:
    seen = seen if seen is not None else set()
    if blocked(path):
        raise ValueError(f"Blocked instruction file: {path}")
    path = path.resolve()
    if path in seen or not path.is_file():
        return []
    if blocked(path):
        raise ValueError(f"Blocked instruction file: {path}")
    if not any(path.is_relative_to(root) for root in config_roots()):
        raise ValueError(f"Instruction include outside configured roots: {path}")
    seen.add(path)
    text = path.read_text(encoding="utf-8-sig")
    documents = [(path, text)]
    for line in text.splitlines():
        match = re.fullmatch(r'@([^\r\n]+\.md)\s*', line.strip())
        if match:
            included = (path.parent / match.group(1).strip()).resolve()
            if not included.is_file():
                raise ValueError(f"Missing instruction include: {included}")
            documents.extend(instruction_documents(included, seen))
    return documents


def root_instructions() -> list[tuple[Path, str]]:
    documents = []
    seen: set[Path] = set()
    for root in reversed(config_roots()):
        for name in ("AGENTS.md", ".github/copilot-instructions.md", "CLAUDE.md"):
            documents.extend(instruction_documents(root / name, seen))
    for path, text in file_instructions(WORKSPACE, directory=True):
        if path not in seen:
            seen.add(path)
            documents.append((path, text))
    return documents


def glob_matches(relative: str, pattern: str) -> bool:
    parts = relative.replace("\\", "/").split("/")
    patterns = pattern.strip().replace("\\", "/").removeprefix("./").split("/")

    def match(path_parts: list[str], glob_parts: list[str]) -> bool:
        if not glob_parts:
            return not path_parts
        if glob_parts[0] == "**":
            return match(path_parts, glob_parts[1:]) or bool(path_parts and match(path_parts[1:], glob_parts))
        return bool(path_parts and fnmatch.fnmatchcase(path_parts[0], glob_parts[0]) and match(path_parts[1:], glob_parts[1:]))

    return match(parts, patterns)


def file_instructions(target: Path, directory: bool = False) -> list[tuple[Path, str]]:
    documents = []
    seen: set[Path] = set()
    for root in reversed(config_roots()):
        relative = target.relative_to(WORKSPACE).as_posix() if target.is_relative_to(WORKSPACE) else target.name
        for path in sorted((root / ".github" / "instructions").glob("**/*.instructions.md")):
            entries = instruction_documents(path)
            if not entries:
                continue
            frontmatter, _body = split_frontmatter(entries[0][1])
            patterns = frontmatter_value(frontmatter, "applyTo").split(",")
            matches = any(pattern.strip() and glob_matches(relative, pattern) for pattern in patterns)
            if directory:
                matches = any(pattern.strip() in {"*", "**", "**/*"} for pattern in patterns)
            if matches:
                for entry_path, text in entries:
                    if entry_path not in seen:
                        seen.add(entry_path)
                        documents.append((entry_path, text))
    if target.is_relative_to(WORKSPACE):
        parent = target if directory else target.parent
        for ancestor in reversed(parent.parents):
            if ancestor != WORKSPACE and ancestor.is_relative_to(WORKSPACE):
                for name in ("AGENTS.md", "CLAUDE.md"):
                    documents.extend(instruction_documents(ancestor / name, seen))
        if parent != WORKSPACE:
            for name in ("AGENTS.md", "CLAUDE.md"):
                documents.extend(instruction_documents(parent / name, seen))
    return documents


def action_instructions(step: Action) -> list[tuple[Path, str]]:
    if step.action == "run_node":
        entries = action_instructions(Action(thought="Check working directory rules", action="run_command", path=step.cwd))
        entries += action_instructions(Action(thought="Check script rules", action="read_file", path=step.path))
        return list(dict(entries).items())
    directory_actions = {"list_dir", "search", "run_command"}
    file_actions = {"read_file", "write_file", "edit_file"}
    if step.action not in directory_actions | file_actions:
        return []
    target = resolve_path(step.path or ".")
    if access_error(target, write=step.action in {"write_file", "edit_file", "run_command"}):
        return []
    return file_instructions(target, directory=step.action in directory_actions and target.is_dir())


def discover_skills() -> list[Skill]:
    found = []
    seen = set()
    for base in config_roots():
        root = base / ".github" / "skills"
        if not root.is_dir():
            continue
        for path in sorted(root.glob("**/SKILL.md")):
            if blocked(path) or not path.resolve().is_relative_to(base):
                continue
            try:
                text = path.read_text(encoding="utf-8")
            except OSError:
                continue
            frontmatter, _body = split_frontmatter(text)
            folder = path.parent
            name = frontmatter_value(frontmatter, "name") or folder.name
            description = frontmatter_value(frontmatter, "description") or prompt_description(text)
            rel_path = str(path.resolve())
            if name.lower() in seen:
                continue
            seen.add(name.lower())
            rel_dir = folder.relative_to(root).as_posix()
            aliases = {
                name,
                folder.name,
                rel_dir,
                rel_path,
                path.relative_to(base).as_posix(),
                name.replace("-", " "),
                name.replace("_", " "),
            }
            found.append(Skill(name, description, rel_path, text, {a.strip() for a in aliases if a.strip()}))
    return found


def _term_pattern(term: str) -> re.Pattern[str]:
    return re.compile(rf"(?<![a-z0-9]){re.escape(term.lower())}(?![a-z0-9])")


def skills_for_texts(texts: list[str]) -> list[Skill]:
    haystack = "\n".join(text for text in texts if text)
    if not haystack.strip():
        return []
    return [skill for skill in discover_skills() if _term_pattern(skill.name).search(haystack.lower())]


def render_skills(skills: list[Skill]) -> str:
    blocks = [
        "Explicitly requested skills (other applicable skills must be loaded with use_skill):"
    ]
    for skill in skills:
        for path, text in instruction_documents(Path(skill.rel_path)):
            _frontmatter, body = split_frontmatter(text)
            blocks.append(f"## {skill.name}: {path}\n{body or text}")
    return "\n\n".join(blocks)


def slash_prompts() -> list[tuple[str, str, str]]:
    found = []
    seen = set()
    for base in config_roots():
        folder = base / ".github" / "prompts"
        if not folder.is_dir():
            continue
        paths = sorted(folder.glob("*.md"), key=lambda path: (not path.name.endswith(".prompt.md"), path.name))
        for path in paths:
            if blocked(path) or not path.resolve().is_relative_to(base):
                continue
            name = path.name.removesuffix(".md").removesuffix(".prompt")
            if name in seen:
                continue
            seen.add(name)
            try:
                text = path.read_text(encoding="utf-8")
            except OSError:
                continue
            found.append((name, prompt_description(text), str(path)))
    return found


def set_workspace(path: Path) -> None:
    global WORKSPACE
    resolved = path.expanduser().resolve()
    if not resolved.is_dir():
        raise ValueError(f"Workspace directory not found: {resolved}")
    WORKSPACE = resolved


def build_system_prompt() -> str:
    parts = [SYSTEM_PROMPT.format(workspace=str(WORKSPACE))]
    parts.append(f"Web tools enabled: {WEB_ENABLED}. Node selection: {NODE_EXECUTABLE or 'workspace node-js.cmd, then node on PATH (>=24 required)'}. Current date: {time.strftime('%Y-%m-%d')}.")
    for path, text in root_instructions():
        parts.append(f"Repository instructions from {path}:\n{text}")
    instructions = []
    for root in config_roots():
        for path in sorted((root / ".github" / "instructions").glob("**/*.instructions.md")):
            documents = instruction_documents(path)
            if documents:
                frontmatter, _body = split_frontmatter(documents[0][1])
                patterns = frontmatter_value(frontmatter, "applyTo") or "on demand"
                instructions.append(f"- {path}: {prompt_description(documents[0][1])} (applyTo: {patterns})")
    if instructions:
        parts.append("Available file instructions. Matching globs are supplied before file actions; read other relevant instructions on demand:\n" + "\n".join(instructions))
    prompts = slash_prompts()
    if prompts:
        lines = [
            "Available slash prompts. Use /name when the user invokes one:"
        ]
        lines.extend(f"- /{name}: {desc} ({path})" for name, desc, path in prompts)
        parts.append("\n".join(lines))
    skills = discover_skills()
    if skills:
        lines = [
            "Available skills in .github/skills/**/SKILL.md. Invoke one when the user prompt, a slash prompt, or CLAUDE.md requires it directly or indirectly:"
        ]
        lines.extend(f"- {skill.name}: {skill.description} ({skill.rel_path})" for skill in skills)
        parts.append("\n".join(lines))
    return "\n\n".join(parts)


def find_skill(name: str) -> Skill | None:
    query = name.strip().lower().removesuffix(".md")
    if not query:
        return None
    for skill in discover_skills():
        aliases = {alias.lower() for alias in skill.aliases}
        aliases.add(skill.name.lower())
        if query in aliases or query == Path(skill.rel_path).parent.name.lower():
            return skill
    return None


def expand_slash(user_text: str) -> str:
    stripped = user_text.strip()
    if not stripped.startswith("/") or stripped.startswith("//"):
        return user_text
    arguments = stripped[1:].split(maxsplit=1)
    if not arguments:
        return user_text
    token = arguments[0]
    rest = arguments[1] if len(arguments) > 1 else ""
    name = token.removesuffix(".md").removesuffix(".prompt")
    if not name or any(sep in name for sep in "/\\"):
        return user_text
    location = next((location for command, _description, location in slash_prompts() if command == name), None)
    if location is None:
        raise ValueError(f"Unknown slash prompt /{name}; expected .github/prompts/{name}.prompt.md or {name}.md")
    path = Path(location)
    documents = instruction_documents(path)
    body = "\n\n".join(f"Instruction document: {document}\n{split_frontmatter(text)[1]}" for document, text in documents)
    extra = rest.strip()
    return (
        f"Slash prompt /{name} from {path}:\n{body}\n\n"
        f"User request:\n{extra or '(no extra arguments)'}"
    )


def designated_instruction(path: Path) -> bool:
    if access_error(path):
        return False
    if path.name in {"AGENTS.md", "CLAUDE.md"}:
        return any(path.resolve().is_relative_to(root) for root in config_roots())
    for root in config_roots():
        if not path.resolve().is_relative_to(root):
            continue
        relative = path.resolve().relative_to(root).as_posix()
        if relative == ".github/copilot-instructions.md":
            return True
        if relative.startswith(".github/instructions/") and path.name.endswith(".instructions.md"):
            return True
        if relative.startswith(".github/prompts/") and path.suffix == ".md":
            return True
        if relative.startswith(".github/skills/") and path.name == "SKILL.md":
            return True
    return False

def resolve_path(raw: str) -> Path:
    candidate = Path(raw).expanduser()
    return Path(os.path.abspath(candidate if candidate.is_absolute() else WORKSPACE / candidate))


def blocked(path: Path, write: bool = False) -> str | None:
    for candidate in (path.absolute(), path.resolve()):
        parts = {part.lower() for part in candidate.parts}
        if any(part in BLOCKED_NAMES or part.startswith(".env.") or part.startswith(".env-")
               or Path(part).suffix in BLOCKED_SUFFIXES for part in parts):
            return f"Refused: {path.name} looks like a secret or credential file."
        if ".git" in parts:
            return "Refused: direct access to .git internals is disabled."
        if "unity3d-projects" in parts and not any(candidate.is_relative_to(root) and "unity3d-projects" in {part.lower() for part in root.parts} for root in ALLOWED_PATHS):
            return "Refused: Unity3D-Projects requires an explicit --allow-path for that project."
        if write and parts.intersection(READ_ONLY_DIRS):
            return "Refused: source and UPM folders are read-only."
    return None


def access_error(path: Path, write: bool = False) -> str | None:
    reason = blocked(path, write)
    if reason:
        return reason
    roots = [WORKSPACE, *ALLOWED_PATHS]
    if not write:
        roots.append(AGENT_HOME)
    if not any(path.resolve().is_relative_to(root.resolve()) for root in roots):
        return "Refused: path outside authorized roots. Restart with --workspace or --allow-path."
    return None


def bounded(text: str, limit: int = MAX_TOOL_CHARS) -> str:
    return text if len(text) <= limit else text[:limit] + "\n(truncated; request a narrower range)"


def file_text(path: Path) -> str:
    if path.stat().st_size > MAX_FILE_BYTES:
        raise ValueError(f"File exceeds {MAX_FILE_BYTES:,} bytes; use a narrower external inspection.")
    return path.read_text(encoding="utf-8-sig")


def rel(path: Path) -> str:
    try:
        return str(path.relative_to(WORKSPACE))
    except ValueError:
        return str(path)


def list_dir(path: str = ".") -> str:
    target = resolve_path(path or ".")
    reason = access_error(target)
    if reason:
        return reason
    if not target.exists():
        return f"Not found: {path}"
    if not target.is_dir():
        return f"Not a directory: {path}"
    entries = []
    for child in sorted(target.iterdir(), key=lambda p: (not p.is_dir(), p.name.lower())):
        if access_error(child):
            continue
        kind = "dir" if child.is_dir() else "file"
        entries.append(f"{kind}\t{rel(child)}")
    return bounded("\n".join(entries)) if entries else "(empty)"


def read_file(path: str, start_line: int = 1, end_line: int | None = None) -> str:
    target = resolve_path(path)
    reason = access_error(target)
    if reason:
        return reason
    if not target.is_file():
        return f"Not a file: {path}"
    start = max(1, int(start_line or 1))
    end = int(end_line) if end_line else start + 399
    if end < start:
        return "end_line must be >= start_line"
    end = min(end, start + 399)
    text = file_text(target)
    lines = text.splitlines()
    selected = lines[start - 1 : end]
    numbered = [f"{i}|{line}" for i, line in enumerate(selected, start=start)]
    header = f"{rel(target)} lines {start}-{min(end, len(lines))} of {len(lines)}"
    body = "\n".join(numbered) if numbered else "(no lines in range)"
    return bounded(header + "\n" + body)


def search(pattern: str, path: str = ".", regex: bool = False) -> str:
    root = resolve_path(path or ".")
    reason = access_error(root)
    if reason:
        return reason
    if not root.exists():
        return f"Not found: {path}"
    flags = 0
    try:
        compiled = re.compile(pattern if regex else re.escape(pattern), flags)
    except re.error as exc:
        return f"Invalid pattern: {exc}"
    def candidates():
        if root.is_file():
            yield root
            return
        for directory, folders, names in os.walk(root, followlinks=False):
            base = Path(directory)
            folders[:] = [name for name in folders if name.lower() not in SKIP_DIRS
                          and not (base / name).is_symlink() and not access_error(base / name)]
            for name in names:
                yield base / name

    hits: list[str] = []
    for file in candidates():
        if access_error(file):
            continue
        if file.suffix.lower() in {".png", ".jpg", ".jpeg", ".gif", ".webp", ".pdf", ".zip", ".exe", ".dll"}:
            continue
        try:
            text = file_text(file)
        except (UnicodeDecodeError, OSError, ValueError):
            continue
        for i, line in enumerate(text.splitlines(), start=1):
            if compiled.search(line):
                hits.append(f"{rel(file)}:{i}:{line[:240]}")
                if len(hits) >= 80:
                    return "\n".join(hits) + "\n(truncated)"
    return "\n".join(hits) if hits else "No matches."


def write_file(path: str, content: str) -> str:
    target = resolve_path(path)
    reason = access_error(target, write=True)
    if reason:
        return reason
    if len(content) > 1_000_000:
        return "Refused: content exceeds 1,000,000 characters."
    if target.exists():
        return "Refused: file already exists; read it and use edit_file for a surgical change."
    target.parent.mkdir(parents=True, exist_ok=True)
    with target.open("x", encoding="utf-8", newline="\n") as stream:
        stream.write(content)
    return f"Created {rel(target)} ({len(content)} chars)"


def edit_file(path: str, old_str: str, new_str: str) -> str:
    target = resolve_path(path)
    reason = access_error(target, write=True)
    if reason:
        return reason
    if not target.is_file():
        return f"Not a file: {path}"
    if old_str == "":
        return "old_str must not be empty."
    if target.stat().st_size > MAX_FILE_BYTES:
        return "Refused: file exceeds the edit size limit."
    original = target.read_bytes()
    text = original.decode("utf-8-sig").replace("\r\n", "\n")
    old_str = old_str.replace("\r\n", "\n")
    new_str = new_str.replace("\r\n", "\n")
    count = text.count(old_str)
    if count == 0:
        return "old_str was not found. Read the file and copy the exact text."
    if count > 1:
        return f"old_str matched {count} times. Include more surrounding context so it is unique."
    updated = text.replace(old_str, new_str, 1)
    if len(updated.encode("utf-8")) > MAX_FILE_BYTES:
        return "Refused: updated file exceeds the size limit."
    newline = "\r\n" if b"\r\n" in original and b"\n" not in original.replace(b"\r\n", b"") else "\n"
    encoding = "utf-8-sig" if original.startswith(b"\xef\xbb\xbf") else "utf-8"
    target.write_text(updated, encoding=encoding, newline=newline)
    return f"Edited {rel(target)}"


def approve_command(command: str, directory: Path) -> bool:
    console.print(Panel(Text(f"Directory: {directory}\n{command}"), title="Unsandboxed command approval"))
    console.print("Commands can access files and the network. Review for secrets or destructive effects.")
    if not sys.stdin.isatty():
        return False
    try:
        return console.input("Run this command? [y/N] ").strip().lower() == "y"
    except (EOFError, KeyboardInterrupt):
        return False


def run_command(command: str, path: str = ".", timeout_seconds: int = 120) -> str:
    directory = resolve_path(path or ".")
    reason = access_error(directory, write=True)
    if reason:
        return reason
    if not command.strip() or not directory.is_dir():
        return "Refused: provide a command and an existing working directory."
    if re.search(r"(?i)(\.env\b|id_rsa|id_ed25519|credentials\.json|NEXUS_API_KEY)", command):
        return "Refused: command references a protected credential."
    shell = shutil.which("pwsh") or shutil.which("powershell")
    if not shell:
        return "Refused: PowerShell is unavailable."
    if not approve_command(command, directory):
        return "Refused: command was not approved; it was not executed."
    environment = {name: value for name, value in os.environ.items()
                   if not re.search(r"(?i)(api.?key|token|secret|password|credential)", name)}
    script = "$ErrorActionPreference = 'Stop'\n" + command + "\nif (-not $?) { exit 1 }; if ($null -ne $LASTEXITCODE) { exit $LASTEXITCODE }"
    with tempfile.TemporaryFile() as output:
        process = subprocess.Popen([shell, "-NoProfile", "-NonInteractive", "-Command", script],
                                   cwd=directory, env=environment, stdin=subprocess.DEVNULL,
                                   stdout=output, stderr=subprocess.STDOUT)
        try:
            process.wait(timeout=max(1, min(timeout_seconds or 120, 600)))
        except (subprocess.TimeoutExpired, KeyboardInterrupt):
            if os.name == "nt":
                subprocess.run(["taskkill", "/PID", str(process.pid), "/T", "/F"],
                               stdout=subprocess.DEVNULL, stderr=subprocess.DEVNULL, check=False)
            process.kill()
            process.wait()
            return "Error: command timed out or was interrupted; process termination requested."
        output.seek(0)
        text = output.read(MAX_TOOL_CHARS + 1).decode("utf-8", errors="replace")
    if API_KEY:
        text = text.replace(API_KEY, "[REDACTED]")
    return f"Exit code: {process.returncode}\n{bounded(text)}"


def powershell_literal(value: str) -> str:
    return "'" + value.replace("'", "''") + "'"


def run_node(path: str, args: list[str], cwd: str = ".", timeout_seconds: int = 120) -> str:
    script = resolve_path(path)
    reason = access_error(script)
    if reason:
        return reason
    if not script.is_file() or script.suffix.lower() not in {".js", ".mjs", ".cjs"}:
        return "Refused: run_node requires an existing .js, .mjs, or .cjs script."
    wrapper = WORKSPACE / "node-js.cmd"
    executable = Path(NODE_EXECUTABLE).expanduser().resolve() if NODE_EXECUTABLE else wrapper if wrapper.is_file() else None
    if executable is None:
        location = shutil.which("node")
        executable = Path(location) if location else None
    if executable is None or not executable.is_file():
        return "Refused: Node unavailable; supply --node-executable or a workspace node-js.cmd."
    command = (
        f"$node = {powershell_literal(str(executable))}\n"
        "$nodeVersion = & $node --version\n"
        "if ($LASTEXITCODE -ne 0) { throw 'Node version check failed' }\n"
        "if (([version](([string]$nodeVersion).Trim().TrimStart('v'))).Major -lt 24) { throw 'Node 24+ required' }\n"
        f"& $node {powershell_literal(str(script))} " + " ".join(powershell_literal(argument) for argument in args)
    )
    return run_command(command, cwd, timeout_seconds)

def web_permission(description: str) -> str | None:
    if not WEB_ENABLED:
        return "Refused: web access disabled by --no-web."
    if (API_KEY and API_KEY in description) or re.search(r"(?i)(NEXUS_API_KEY|password=|api[_-]?key=|token=|secret=)", description):
        return "Refused: possible credential in web request."
    if not approve_web(description):
        return "Refused: web request was not approved; nothing was sent."
    return None


def public_url(url: str) -> str:
    parts = urlsplit(url)
    if parts.scheme not in {"https", "http"} or not parts.hostname or parts.username or parts.password:
        raise ValueError("Only public HTTP(S) URLs without embedded credentials are supported.")
    if parts.port not in {None, 80, 443}:
        raise ValueError("Only public web ports 80 and 443 are supported.")
    addresses = socket.getaddrinfo(parts.hostname, parts.port or (443 if parts.scheme == "https" else 80), type=socket.SOCK_STREAM)
    if not addresses or any(not ipaddress.ip_address(address[4][0]).is_global for address in addresses):
        raise ValueError("Private, loopback, link-local, and reserved web destinations are blocked.")
    return url


def web_search(query: str, max_results: int = 5) -> str:
    if not query.strip() or len(query) > 1000:
        return "Invalid: provide a public search query of 1-1000 characters."
    reason = web_permission("Search: " + query)
    if reason:
        return reason
    from ddgs import DDGS

    rows = DDGS(timeout=20).text(query, max_results=max(1, min(max_results or 5, 10)), backend="duckduckgo")
    results = [{"title": row.get("title", ""), "url": row.get("href", ""), "snippet": row.get("body", "")} for row in rows]
    if not results:
        return "No web results. Do not invent sources; try a more specific public query."
    return bounded(json.dumps({"query": query, "results": results}, ensure_ascii=True, indent=2))


def fetch_webpage(url: str) -> str:
    import httpx
    from bs4 import BeautifulSoup

    current = url
    for _redirect in range(6):
        reason = web_permission("Fetch: " + current)
        if reason:
            return reason
        public_url(current)
        with httpx.Client(timeout=20, follow_redirects=False, trust_env=False) as client:
            with client.stream("GET", current, headers={"User-Agent": "NexusCodingAgent/1.0", "Accept": "text/html,text/plain,application/json"}) as response:
                if response.is_redirect:
                    location = response.headers.get("location")
                    if not location:
                        return "Error: redirect has no Location header."
                    current = urljoin(current, location)
                    continue
                response.raise_for_status()
                content_type = response.headers.get("content-type", "").split(";", 1)[0].lower()
                if content_type not in {"text/html", "application/xhtml+xml", "text/plain", "application/json"}:
                    return "Refused: only HTML, plain text, and JSON pages are supported."
                chunks = bytearray()
                deadline = time.monotonic() + 30
                for chunk in response.iter_bytes(chunk_size=65536):
                    chunks.extend(chunk)
                    if len(chunks) > MAX_FILE_BYTES or time.monotonic() > deadline:
                        return "Refused: web response exceeded size or duration limit."
                text = chunks.decode(response.encoding or "utf-8", errors="replace")
        title = ""
        if content_type in {"text/html", "application/xhtml+xml"}:
            soup = BeautifulSoup(text, "html.parser")
            title = soup.title.get_text(" ", strip=True) if soup.title else ""
            for element in soup(["script", "style", "noscript", "template"]):
                element.decompose()
            text = (soup.find("main") or soup.find("article") or soup).get_text("\n", strip=True)
        return f"URL: {current}\nTitle: {title}\n\n{bounded(text)}"
    return "Error: too many web redirects."


def run_action(step: Action) -> str:
    try:
        if step.action == "list_dir":
            return list_dir(step.path or ".")
        if step.action == "read_file":
            end = step.end_line if step.end_line else None
            return read_file(step.path, step.start_line or 1, end)
        if step.action == "search":
            return search(step.pattern, step.path or ".", step.regex)
        if step.action == "write_file":
            return write_file(step.path, step.content)
        if step.action == "edit_file":
            return edit_file(step.path, step.old_str, step.new_str)
        if step.action == "run_command":
            return run_command(step.command, step.path or ".", step.timeout_seconds)
        if step.action == "run_node":
            return run_node(step.path, step.args, step.cwd, step.timeout_seconds)
        if step.action == "web_search":
            return web_search(step.query, step.max_results)
        if step.action == "fetch_webpage":
            return fetch_webpage(step.url)
        if step.action == "use_instructions":
            path = resolve_path(step.path)
            if not designated_instruction(path) or not path.is_file():
                return "Refused: use_instructions requires an existing designated instruction document."
            return "\n\n".join(f"Repository instructions from {document}:\n{text}" for document, text in instruction_documents(path))
        if step.action == "ask_user":
            if not step.reply.strip():
                return "Invalid: ask_user requires a question in reply."
            if not sys.stdin.isatty():
                return "Refused: no interactive input is available. Stop and report the missing information."
            console.print(Text(step.reply))
            return "User answer: " + console.input("> ")
        if step.action == "use_skill":
            skill = find_skill(step.skill or step.path)
            if skill is None:
                known = ", ".join(item.name for item in discover_skills()) or "(none)"
                return f"No skill named {step.skill or step.path}. Known skills: {known}"
            return render_skills([skill])
        if step.action == "done":
            return step.reply or "Done."
        return f"Unknown action: {step.action}"
    except TypeError as exc:
        return f"Invalid arguments for {step.action}: {exc}"
    except Exception as exc:
        return f"{type(exc).__name__}: {exc}"

def show_tool_result(action: str, target: str, result: str) -> None:
    if not result:
        return
    if action == "list_dir":
        for line in result.splitlines():
            kind, _, name = line.partition("\t")
            if kind == "dir":
                console.print(f"[bold cyan]{name}/[/bold cyan]")
            elif kind == "file":
                console.print(name)
            else:
                console.print(line, style="dim")
        return
    if action == "read_file" and " lines " in result.splitlines()[0]:
        header, _, body = result.partition("\n")
        console.print(f"[dim]{header}[/dim]")
        if not body or body.startswith("(no lines"):
            console.print(body, style="dim")
            return
        code_lines = []
        start = 1
        for index, line in enumerate(body.splitlines()):
            number, sep, code = line.partition("|")
            if sep and number.isdigit():
                if index == 0:
                    start = int(number)
                code_lines.append(code)
            else:
                code_lines.append(line)
        show_code(header.split(" ", 1)[0], "\n".join(code_lines), start)
        return
    if action == "search":
        lines = result.splitlines()
        for line in lines[:40]:
            path, sep, rest = line.partition(":")
            if sep and rest[:1].isdigit():
                number, _, code = rest.partition(":")
                console.print(f"[cyan]{path}[/cyan]:[yellow]{number}[/yellow]: {code}")
            else:
                console.print(line, style="dim")
        if len(lines) > 40:
            console.print(f"[dim]… {len(lines) - 40} more matches[/dim]")
        return
    if result.startswith(("Created ", "Overwrote ", "Edited ")):
        console.print(f"[bold green]{result}[/bold green]")
        return
    if result.startswith(("Refused", "Not ", "No ", "old_str", "Invalid", "Error", "Unknown")):
        preview = result if len(result) < 800 else result[:800] + "…"
        console.print(preview, style="red")
        return
    preview = result if len(result) < 800 else result[:800] + "…"
    console.print(preview, style="dim")

def keyboard_text(text: str) -> str:
    return "".join(character if character in "\t\n\r" or " " <= character <= "~"
                   else character.encode("unicode_escape").decode("ascii") for character in text)


def markdown_block(text: str, language: str = "text") -> str:
    longest = max((len(match.group()) for match in re.finditer(r"`+", text)), default=0)
    fence = "`" * max(3, longest + 1)
    return f"{fence}{language}\n{text}\n{fence}"


def session_slug(text: str) -> str:
    words = re.findall(r"[a-z0-9]+", text.lower())
    words = [word for word in words if word not in {"a", "an", "the", "please", "this", "that", "to", "of", "for", "and", "in"}]
    return "-".join(word[:40] for word in (words + ["agent", "session"])[:2])


class Transcript:
    def __init__(self, session_name: str = "") -> None:
        self.parts: list[str] = []
        self.workspace = WORKSPACE
        self.started = int(time.time())
        self.identifier = uuid.uuid4().hex[:8]
        self.label = session_slug(session_name) if session_name else ""
        self.folder_name = f"session-{self.label}" if session_name else ""

    def add(self, title: str, body: str) -> None:
        if not body:
            return
        if title == "Request" and not self.label:
            self.label = session_slug(body)
        self.parts.append(f"## {len(self.parts) + 1}. {title}\n\n{body}")

    def add_action(self, step: Action) -> None:
        fields = step.model_dump(exclude={"thought", "content", "old_str", "new_str"}, exclude_defaults=True)
        body = markdown_block(json.dumps(fields, ensure_ascii=True, indent=2), "json")
        if step.action == "write_file":
            body += "\n\n### Proposed File Content\n\n" + markdown_block(step.content, lexer_for(step.path))
        elif step.action == "edit_file":
            body += "\n\n### Old Text\n\n" + markdown_block(step.old_str, lexer_for(step.path))
            body += "\n\n### Proposed New Text\n\n" + markdown_block(step.new_str, lexer_for(step.path))
        self.add("Action: " + step.action, body)

    def text(self) -> str:
        text = (
            f"# Chat Session: {self.label or 'agent-session'}\n\n"
            f"Started (epoch seconds): {self.started}\n\nWorkspace: {self.workspace}\n\n"
            "Progress entries are model-provided summaries, not private reasoning.\n"
            "Non-keyboard characters use visible Unicode/control escapes; source files are unchanged.\n"
            "Action content is a proposal; the following tool result records whether it ran.\n"
            "Tool results include the text returned to the agent, which may already be truncated.\n\n"
            + "\n\n".join(self.parts) + "\n"
        )
        if API_KEY:
            text = text.replace(API_KEY, "[REDACTED]")
        return keyboard_text(text)

    def save(self) -> tuple[Path, str]:
        if not self.parts:
            raise ValueError("No conversation to export yet.")
        if not self.folder_name:
            self.folder_name = f"{self.started}-{self.label or 'agent-session'}-{self.identifier}"
        directory = self.workspace / "chat-history" / self.folder_name
        reason = access_error(directory, write=True)
        if reason:
            raise ValueError(reason)
        directory.mkdir(parents=True, exist_ok=True)
        text = self.text()
        stem = f"{int(time.time())}-{self.label or 'agent-session'}"
        counter = 1
        while True:
            suffix = "" if counter == 1 else f"-{counter}"
            path = directory / f"{stem}{suffix}.md"
            reason = access_error(path, write=True)
            if reason:
                raise ValueError(reason)
            try:
                with path.open("x", encoding="utf-8", newline="\n") as output:
                    output.write(text)
                return path, text
            except FileExistsError:
                counter += 1


def export_conversation(transcript: Transcript) -> Path:
    path, text = transcript.save()
    console.print(Text(f"Saved conversation: {path}", style="green"))
    try:

def copy_to_clipboard(text: str) -> None:
    shell = shutil.which("pwsh") or shutil.which("powershell")
    if not shell:
        raise RuntimeError("PowerShell is unavailable; use the saved Markdown file.")
    subprocess.run(
        [
            shell,
            "-NoProfile",
            "-NonInteractive",
            "-Command",
            "$ErrorActionPreference = 'Stop'; Set-Clipboard -Value ([Console]::In.ReadToEnd())",
        ],
        input=keyboard_text(text).encode("ascii"),
        check=True,
        timeout=15,
    )


class Agent:
    def __init__(self, model: str, effort: str, show_reasoning: bool = False, max_steps: int = 100, session_name: str = ""):
        self.client = create_client()
        self.model = model
        self.effort = effort
        self.show_reasoning = show_reasoning
        self.max_steps = max_steps
        self.messages: list[dict] = [{"role": "system", "content": build_system_prompt()}]
        self.transcript = Transcript(session_name)

    def complete(self, messages: list[dict]):
        if MAX_COMPLETION_TOKENS <= 0 or MAX_CONTEXT_TOKENS <= MAX_COMPLETION_TOKENS:
            raise ValueError("Token limits must be positive, with context larger than completion reserve.")
        payload = json.dumps({"messages": messages, "schema": Action.model_json_schema()}, ensure_ascii=False)
        upper_bound = len(payload.encode("utf-8")) + 4096
        if upper_bound + MAX_COMPLETION_TOKENS > MAX_CONTEXT_TOKENS:
            raise ValueError("Context budget reached (conservative UTF-8 byte estimate plus completion reserve). Start a fresh session; history was not truncated.")
        return self.client.beta.chat.completions.parse(
            model=self.model,
            messages=messages,
            response_format=Action,
            reasoning_effort=self.effort,
            max_completion_tokens=MAX_COMPLETION_TOKENS,
        )

    def ask(self, user_text: str) -> str:
        self.transcript.add("Request", user_text)
        self.messages[0] = {"role": "system", "content": build_system_prompt()}
        expanded = expand_slash(user_text)
        skills = skills_for_texts([user_text])
        loaded_instructions = dict(root_instructions())
        for skill in skills:
            show_skill(skill.name)
        if skills:
            expanded += "\n\n" + render_skills(skills)
        if expanded != user_text:
            self.transcript.add("Expanded Request Context", markdown_block(expanded, "markdown"))
        self.messages.append({"role": "user", "content": expanded})
        for _step_index in range(self.max_steps):
            self.messages[0] = {"role": "system", "content": build_system_prompt()}
            show_thinking()
            response = self.complete(self.messages)
            message = response.choices[0].message
            if message.refusal:
                self.messages.append({"role": "assistant", "content": message.refusal})
                self.transcript.add("Response", message.refusal)
                return message.refusal
            step = message.parsed
            if step is None:
                raise ValueError("Model returned no valid Action; no action was executed.")
            thought = (step.thought or "").strip()
            if thought:
                self.transcript.add("Progress", thought)
                if self.show_reasoning:
                    show_thought(thought)
            self.messages.append({"role": "assistant", "content": step.model_dump_json()})
            if step.action == "done":
                if not step.reply.strip():
                    raise ValueError("The done action requires a nonempty reply.")
                reply = step.reply
                self.transcript.add("Response", reply)
                return reply
            self.transcript.add_action(step)
            target = {"use_skill": step.skill, "run_command": step.command,
                      "web_search": step.query, "fetch_webpage": step.url}.get(step.action, step.path)
            show_tool(step.action, target)
            pending_instructions = [(path, text) for path, text in action_instructions(step)
                                    if loaded_instructions.get(path) != text]
            if pending_instructions:
                loaded_instructions.update(pending_instructions)
                result = "Action not executed. Read these applicable instructions, then reconsider and retry:\n\n"
                result += "\n\n".join(f"Instructions from {path}:\n{text}" for path, text in pending_instructions)
            else:
                result = run_action(step)
            instruction_result = bool(pending_instructions) or step.action in {"use_skill", "use_instructions"}
            if step.action == "read_file" and designated_instruction(resolve_path(step.path)):
                instruction_result = True
            self.transcript.add("Tool Result: " + step.action, markdown_block(result))
            show_tool_result(step.action, target, result)
            if step.action == "write_file" and result.startswith(("Created ", "Overwrote ")):
                show_code(step.path, step.content)
            elif step.action == "edit_file" and result.startswith("Edited "):
                console.print("[red]- old[/red]")
                show_code(step.path, step.old_str)
                console.print("[green]+ new[/green]")
                show_code(step.path, step.new_str)
            self.messages.append(
                {
                    "role": "user",
                    "content": (
                        "TOOL_RESULT:\n"
                        + json.dumps({"action": step.action,
                                      "kind": "repository_instructions" if instruction_result
                                      else "user_answer" if step.action == "ask_user" else "untrusted_data",
                                  "result": result}, ensure_ascii=True)
                        + "\nContinue with the next action, or action=done and a short reply."
                    ),
                }
            )
        reply = f"Stopped after {self.max_steps} actions; task may be incomplete. Review the results and ask to continue, or start a fresh session."
        self.transcript.add("Response", reply)
        self.messages.append({"role": "assistant", "content": reply})
        raise RuntimeError(reply)

def render_3d(
    text: str,
    face: str = "#",
    shadow: str = ".",
    depth: int = 2,
    scale: int = 2,
) -> str:
    """Return *text* as bold bitmap letters with a down-right 3D shadow."""
    glyphs = [FONT.get(character, FONT["?"]) for character in text.upper()]
    face_pixels: set[tuple[int, int]] = set()
    cursor = 0

    for glyph in glyphs:
        for y, row in enumerate(glyph):
            for x, pixel in enumerate(row):
                if pixel == "1":
                    # Expanding in both directions gives the letters a bold weight.
                    for scaled_y in range(scale):
                        for scaled_x in range(scale):
                            face_pixels.add(
                                (
                                    (cursor + x) * scale + scaled_x,
                                    y * scale + scaled_y,
                                )
                            )
        cursor += len(glyph[0]) + 1

    if not face_pixels:
        return ""

    width = (cursor - 1) * scale + depth
    height = 7 * scale + depth
    canvas = [[" " for _ in range(width)] for _ in range(height)]

    # Draw every offset layer, from back to front, to make a solid extrusion.
    for offset in range(depth, 0, -1):
        for x, y in face_pixels:
            canvas[y + offset][x + offset] = shadow

    for x, y in face_pixels:
        canvas[y][x] = face

    return "\n".join("".join(row).rstrip() for row in canvas).rstrip()


def single_character(value: str) -> str:
    """Validate argparse options which must contain one character."""
    if len(value) != 1 or value.isspace():
        raise argparse.ArgumentTypeError("must be exactly one non-space character")
    return value

Why is GPT-5.1 slower by default?¶
gpt-5, gpt-5.1, and gpt-5.2 are reasoning models — before producing a final answer, they perform an internal chain-of-thought to break down the problem, evaluate options, and verify their output. This extra "thinking" step improves accuracy and reliability, especially on complex tasks, but comes at the cost of higher latency and token usage.

The tradeoff is straightforward: lowering reasoning effort reduces latency and token cost, but also reduces the model's accuracy on complex tasks. For simple tasks (classification, extraction, chat) this has little to no impact. For multi-step reasoning or analysis, you may notice a drop in quality.

Tune reasoning_effort to match your task (see Reasoning):

Task Complexity	reasoning_effort	Behavior
Complex analysis, multi-step reasoning	high	Full reasoning — highest accuracy, highest latency
General-purpose tasks	medium (default)	Balanced reasoning
Simple chat, extraction, classification	low / minimal	Reduced reasoning — faster, slightly less depth
Latency-critical (autocomplete, routing)	none	No reasoning — behaves like a traditional model, fastest response times
Tip

With reasoning_effort set to none, gpt-5.1 is often faster than gpt-5-nano while being more capable. This makes it a strong single-model choice across a wide range of use cases.

Function Calling / Tool Use¶
Function calling (also known as tool calling) provides a powerful and flexible way for OpenAI models to interface with external systems and access data outside their training data. This guide shows how you can connect a model to data and actions provided by your application. [OpenAI Function Calling Guide]

Function Calling Flow

First let's define a helper function that help us validate tool invocations to prevent hallucination:


import inspect


def check_args(function, args):
    sig = inspect.signature(function)
    params = sig.parameters

    # Check if there are extra arguments
    for name in args:
        if name not in params:
            return False
    # Check if the required arguments are provided
    for name, param in params.items():
        if param.default is param.empty and name not in args:
            return False

    return True

    model="<model-id>",
    messages=messages,
    tools=tools,

    # extend message with model's response
    messages.append(response_message)

    for tool_call in tool_calls:
        fn_name = tool_call.function.name
        args = tool_call.function.arguments
        fn = {
            "get_current_weather": get_current_weather,
        }.get(fn_name)

        # 1. Validation
        if fn is None:
            raise ValueError(f"Function {fn_name} invoked by model is not a valid function")
        if not check_args(fn, args):
            raise ValueError(f"Function {fn_name} invoked by model has invalid arguments: {args}")

        # 2. Call the tool
        function_response = fn(**args)

        # 3. Extend the messages with function response
        messages.append(
            {
                "tool_call_id": tool_call.id,
                "role": "tool",
                "name": fn_name,
                "content": function_response,
            }
        )

    # 4. Get final response from model
    final_response = client.chat.completions.create(
        model="<model-id>",
        messages=messages,
    )

def process_response_stream_with_tool_calls(stream):
    tool_calls = []
    delta = None

    for chunk in stream:
        delta = chunk.choices[0].delta if chunk.choices and chunk.choices[0].delta is not None else None

        if delta and delta.content:
            print(delta.content, end="", flush=True)

        elif delta and delta.tool_calls:
            tc_chunk_list = delta.tool_calls
            for tc_chunk in tc_chunk_list:
                if len(tool_calls) <= tc_chunk.index:
                    tool_calls.append({"id": "", "type": "function", "function": {"name": "", "arguments": ""}})
                tc = tool_calls[tc_chunk.index]

                if tc_chunk.id:
                    tc["id"] += tc_chunk.id
                if tc_chunk.function.name:
                    tc["function"]["name"] += tc_chunk.function.name
                if tc_chunk.function.arguments:
                    tc["function"]["arguments"] += tc_chunk.function.arguments

    def setUp(self):
        self.temporary = tempfile.TemporaryDirectory()
        self.addCleanup(self.temporary.cleanup)
        self.root = Path(self.temporary.name)
        self.workspace = self.root / "workspace"
        self.home = self.root / "home"
        self.workspace.mkdir()
        self.home.mkdir()
        for name, value in (("WORKSPACE", self.workspace), ("AGENT_HOME", self.home), ("ALLOWED_PATHS", []), ("NODE_EXECUTABLE", ""), ("WEB_ENABLED", True)):
            patcher = patch.object(agent, name, value)
            patcher.start()
            self.addCleanup(patcher.stop)

    def write(self, relative, content, root=None):
        path = (root or self.workspace) / relative
        path.parent.mkdir(parents=True, exist_ok=True)
        path.write_text(content, encoding="utf-8")
        return path
