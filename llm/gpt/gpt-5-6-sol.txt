SYSTEM_PROMPT = """You are a coding agent with high-effort reasoning. You create and edit files in the workspace by returning exactly one action.

This gateway does not allow native function tools together with reasoning, so every response must be one Action object. Put the reasoning for this step in thought. That text is shown to the user while the step runs, so make it specific and concise.

Actions:
- list_dir: set path (use "." for the workspace root).
- read_file: set path, start_line, end_line. end_line 0 means a 400-line window.
- search: set pattern, path, and regex.
- write_file: set path and content. Creates parent directories. Use this to create files.
- edit_file: set path, old_str, and new_str. old_str must match exactly once.
- done: set reply to the final answer for the user. Do this only after the requested file changes exist.

Rules:
- If CLAUDE.md exists at the workspace root, it is injected into every request below. Follow it.
- Slash prompts live in .github/prompts/*.md. If the user starts with /name, that file is the instruction for this request. If a listed slash prompt matches the task, read and follow it before editing files.
- Skills live in .github/skills/**/SKILL.md. If the user prompt, a slash prompt, or CLAUDE.md requires a skill directly or indirectly, follow that skill before editing files. Matched skills are injected into the request. If another listed skill applies, use action use_skill before editing.
- Inspect with list_dir, search, and read_file before editing existing files.
- Prefer edit_file for surgical changes. Use write_file for new files.
- Stay inside the workspace. Never read or write secrets (.env, keys, credentials).
- Fill unused string fields with "" and unused integers with 0. Set regex false unless needed. thought must not be empty.
- After a TOOL_RESULT message, choose the next action. Do not claim a file changed until a tool result says it did.

class Action(BaseModel):
    thought: str = Field(description="Reasoning for this step. Shown to the user.")
    action: Literal["list_dir", "read_file", "search", "write_file", "edit_file", "use_skill", "done"]
    path: str = ""
    skill: str = Field(default="", description="Skill name for use_skill.")
    start_line: int = 1
    end_line: int = 0
    pattern: str = ""
    regex: bool = False
    content: str = ""
    old_str: str = ""
    new_str: str = ""
    reply: str = Field(default="", description="Final answer. Required when action is done.")


def split_frontmatter(text: str) -> tuple[str, str]:
    if not text.startswith("---"):
        return "", text
    end = text.find("\n---", 3)
    if end == -1:
        return "", text
    return text[3:end], text[end + 4 :].lstrip("\n")


def prompt_description(text: str) -> str:
    frontmatter, body = split_frontmatter(text)
    for line in frontmatter.splitlines():
        if line.lower().startswith("description:"):
            return line.split(":", 1)[1].strip().strip("\"'")
    for line in body.splitlines():
        stripped = line.strip().lstrip("#").strip()
        if stripped:
            return stripped[:160]
    return "(no description)"


class Skill:
    def __init__(self, name: str, description: str, rel_path: str, text: str, aliases: set[str]):
        self.name = name
        self.description = description
        self.rel_path = rel_path
        self.text = text
        self.aliases = aliases


def frontmatter_value(frontmatter: str, key: str) -> str:
    prefix = key.lower() + ":"
    for line in frontmatter.splitlines():
        if line.lower().startswith(prefix):
            return line.split(":", 1)[1].strip().strip("\"'")
    return ""


def discover_skills() -> list[Skill]:
    root = WORKSPACE / ".github" / "skills"
    if not root.is_dir():
        return []
    found = []
    for path in sorted(root.glob("**/SKILL.md")):
        try:
            text = path.read_text(encoding="utf-8")
        except OSError:
            continue
        frontmatter, _body = split_frontmatter(text)
        folder = path.parent
        name = frontmatter_value(frontmatter, "name") or folder.name
        description = frontmatter_value(frontmatter, "description") or prompt_description(text)
        rel_path = path.relative_to(WORKSPACE).as_posix()
        rel_dir = folder.relative_to(root).as_posix()
        aliases = {
            name,
            folder.name,
            rel_dir,
            rel_path,
            name.replace("-", " "),
            name.replace("_", " "),
        }
        found.append(Skill(name, description, rel_path, text, {a.strip() for a in aliases if a.strip()}))
    return found


def _term_pattern(term: str) -> re.Pattern[str]:
    return re.compile(rf"(?<![a-z0-9]){re.escape(term.lower())}(?![a-z0-9])")

def skill_mentioned(haystack: str, skill: Skill) -> bool:
    text = haystack.lower()
    for alias in skill.aliases:
        alias = alias.lower()
        if len(alias) < 2:
            continue
        if " " in alias or "/" in alias:
            if alias in text:
                return True
            continue
        if _term_pattern(alias).search(text):
            return True
    words = re.findall(r"[a-z0-9]+", skill.description.lower())
    for index in range(len(words) - 2):
        phrase = " ".join(words[index : index + 3])
        if phrase in text and any(word not in INDIRECT_STOPWORDS and len(word) > 3 for word in words[index : index + 3]):
            return True
    terms = [word for word in dict.fromkeys(words) if len(word) > 3 and word not in INDIRECT_STOPWORDS]
    if len(terms) < 2:
        return False
    hits = [term for term in terms if _term_pattern(term).search(text)]
    return len(hits) >= 2 and len(hits) / len(terms) >= 0.5


def skills_for_texts(texts: list[str]) -> list[Skill]:
    haystack = "\n".join(text for text in texts if text)
    if not haystack.strip():
        return []
    return [skill for skill in discover_skills() if skill_mentioned(haystack, skill)]


def render_skills(skills: list[Skill]) -> str:
    blocks = [
        "Required skills (mentioned directly or indirectly by the user prompt, slash prompt, or CLAUDE.md; follow them before other work):"
    ]
    for skill in skills:
        _frontmatter, body = split_frontmatter(skill.text)
        blocks.append(f"## {skill.name} ({skill.rel_path})\n{body or skill.text}")
    return "\n\n".join(blocks)


def slash_prompts() -> list[tuple[str, str, str]]:
    folder = WORKSPACE / ".github" / "prompts"
    if not folder.is_dir():
        return []
    found = []
    for path in sorted(folder.glob("*.md")):
        try:
            text = path.read_text(encoding="utf-8")
        except OSError:
            continue
        found.append((path.stem, prompt_description(text), f".github/prompts/{path.name}"))
    return found


def build_system_prompt() -> str:
    parts = [SYSTEM_PROMPT.format(workspace=str(WORKSPACE))]
    claude = WORKSPACE / "CLAUDE.md"
    if claude.is_file():
        parts.append("CLAUDE.md (injected into every request; follow it):\n" + claude.read_text(encoding="utf-8"))
    prompts = slash_prompts()
    if prompts:
        lines = [
            "Available slash prompts. Use /name when the user invokes one, and read the file when a prompt matches the task:"
        ]
        lines.extend(f"- /{name}: {desc} ({path})" for name, desc, path in prompts)
        parts.append("\n".join(lines))
    skills = discover_skills()
    if skills:
        lines = [
            "Available skills in .github/skills/**/SKILL.md. Invoke one when the user prompt, a slash prompt, or CLAUDE.md requires it directly or indirectly:"
        ]
        lines.extend(f"- {skill.name}: {skill.description} ({skill.rel_path})" for skill in skills)
        parts.append("\n".join(lines))
    return "\n\n".join(parts)


def find_skill(name: str) -> Skill | None:
    query = name.strip().lower().removesuffix(".md")
    if not query:
        return None
    for skill in discover_skills():
        aliases = {alias.lower() for alias in skill.aliases}
        aliases.add(skill.name.lower())
        if query in aliases or query == Path(skill.rel_path).parent.name.lower():
            return skill
    return None


def expand_slash(user_text: str) -> str:
    stripped = user_text.strip()
    if not stripped.startswith("/") or stripped.startswith("//"):
        return user_text
    token, _, rest = stripped[1:].partition(" ")
    name = token[:-3] if token.lower().endswith(".md") else token
    if not name or any(sep in name for sep in "/\\"):
        return user_text
    path = WORKSPACE / ".github" / "prompts" / f"{name}.md"
    if not path.is_file():
        return user_text + f"\n\n(No slash prompt at .github/prompts/{name}.md)"
    _, body = split_frontmatter(path.read_text(encoding="utf-8"))
    extra = rest.strip()
    return (
        f"Slash prompt /{name} from .github/prompts/{name}.md:\n{body}\n\n"
        f"User request:\n{extra or '(no extra arguments)'}"
    )

def resolve_path(raw: str) -> Path:
    candidate = Path(raw)
    path = candidate.resolve() if candidate.is_absolute() else (WORKSPACE / candidate).resolve()
    if path != WORKSPACE and WORKSPACE not in path.parents:
        raise ValueError(f"Path is outside the workspace: {raw}")
    return path


def blocked(path: Path) -> str | None:
    name = path.name.lower()
    if name in BLOCKED_NAMES or path.suffix.lower() in BLOCKED_SUFFIXES:
        return f"Refused: {path.name} looks like a secret or credential file."
    return None


def rel(path: Path) -> str:
    try:
        return str(path.relative_to(WORKSPACE))
    except ValueError:
        return str(path)


def list_dir(path: str = ".") -> str:
    target = resolve_path(path or ".")
    if not target.exists():
        return f"Not found: {path}"
    if not target.is_dir():
        return f"Not a directory: {path}"
    entries = []
    for child in sorted(target.iterdir(), key=lambda p: (not p.is_dir(), p.name.lower())):
        if child.name in BLOCKED_NAMES:
            continue
        kind = "dir" if child.is_dir() else "file"
        entries.append(f"{kind}\t{rel(child)}")
    return "\n".join(entries) if entries else "(empty)"


def read_file(path: str, start_line: int = 1, end_line: int | None = None) -> str:
    target = resolve_path(path)
    reason = blocked(target)
    if reason:
        return reason
    if not target.is_file():
        return f"Not a file: {path}"
    start = max(1, int(start_line or 1))
    end = int(end_line) if end_line else start + 399
    if end < start:
        return "end_line must be >= start_line"
    text = target.read_text(encoding="utf-8")
    lines = text.splitlines()
    selected = lines[start - 1 : end]
    numbered = [f"{i}|{line}" for i, line in enumerate(selected, start=start)]
    header = f"{rel(target)} lines {start}-{min(end, len(lines))} of {len(lines)}"
    body = "\n".join(numbered) if numbered else "(no lines in range)"
    return header + "\n" + body


def search(pattern: str, path: str = ".", regex: bool = False) -> str:
    import re

    root = resolve_path(path or ".")
    if not root.exists():
        return f"Not found: {path}"
    flags = 0
    try:
        compiled = re.compile(pattern if regex else re.escape(pattern), flags)
    except re.error as exc:
        return f"Invalid pattern: {exc}"
    files = [root] if root.is_file() else [p for p in root.rglob("*") if p.is_file()]
    hits: list[str] = []
    for file in files:
        if blocked(file) or any(part in BLOCKED_NAMES for part in file.parts):
            continue
        if file.suffix.lower() in {".png", ".jpg", ".jpeg", ".gif", ".webp", ".pdf", ".zip", ".exe", ".dll"}:
            continue
        try:
            text = file.read_text(encoding="utf-8")
        except (UnicodeDecodeError, OSError):
            continue
        for i, line in enumerate(text.splitlines(), start=1):
            if compiled.search(line):
                hits.append(f"{rel(file)}:{i}:{line[:240]}")
                if len(hits) >= 80:
                    return "\n".join(hits) + "\n(truncated)"
    return "\n".join(hits) if hits else "No matches."


def write_file(path: str, content: str) -> str:
    target = resolve_path(path)
    reason = blocked(target)
    if reason:
        return reason
    if len(content) > 1_000_000:
        return "Refused: content exceeds 1,000,000 characters."
    target.parent.mkdir(parents=True, exist_ok=True)
    existed = target.exists()
    target.write_text(content, encoding="utf-8", newline="\n")
    action = "Overwrote" if existed else "Created"
    return f"{action} {rel(target)} ({len(content)} chars)"


def edit_file(path: str, old_str: str, new_str: str) -> str:
    target = resolve_path(path)
    reason = blocked(target)
    if reason:
        return reason
    if not target.is_file():
        return f"Not a file: {path}"
    if old_str == "":
        return "old_str must not be empty."
    text = target.read_text(encoding="utf-8")
    count = text.count(old_str)
    if count == 0:
        return "old_str was not found. Read the file and copy the exact text."
    if count > 1:
        return f"old_str matched {count} times. Include more surrounding context so it is unique."
    updated = text.replace(old_str, new_str, 1)
    target.write_text(updated, encoding="utf-8", newline="\n")
    return f"Edited {rel(target)}"


def run_action(step: Action) -> str:
    try:
        if step.action == "list_dir":
            return list_dir(step.path or ".")
        if step.action == "read_file":
            end = step.end_line if step.end_line else None
            return read_file(step.path, step.start_line or 1, end)
        if step.action == "search":
            return search(step.pattern, step.path or ".", step.regex)
        if step.action == "write_file":
            return write_file(step.path, step.content)
        if step.action == "edit_file":
            return edit_file(step.path, step.old_str, step.new_str)
        if step.action == "use_skill":
            skill = find_skill(step.skill or step.path)
            if skill is None:
                known = ", ".join(item.name for item in discover_skills()) or "(none)"
                return f"No skill named {step.skill or step.path}. Known skills: {known}"
            return f"Follow this skill ({skill.rel_path}):\n{skill.text}"
        if step.action == "done":
            return step.reply or "Done."
        return f"Unknown action: {step.action}"
    except TypeError as exc:
        return f"Invalid arguments for {step.action}: {exc}"
    except Exception as exc:
        return f"{type(exc).__name__}: {exc}"


def lexer_for(path: str) -> str:
    return LEXERS.get(Path(path).suffix.lower(), "text")


def syntax_block(code: str, path: str, start_line: int = 1) -> Syntax:
    return Syntax(
        code,
        lexer_for(path),
        theme="monokai",
        line_numbers=True,
        start_line=start_line,
        background_color="default",
        word_wrap=True,
    )

def show_code(path: str, code: str, start_line: int = 1) -> None:
    lines = code.splitlines()
    shown = "\n".join(lines[:80])
    if shown:
        console.print(syntax_block(shown, path, start_line))
    if len(lines) > 80:
        console.print(f"[dim]… {len(lines) - 80} more lines[/dim]")


def show_tool_result(action: str, target: str, result: str) -> None:
    if not result:
        return
    if action == "list_dir":
        for line in result.splitlines():
            kind, _, name = line.partition("\t")
            if kind == "dir":
                console.print(f"[bold cyan]{name}/[/bold cyan]")
            elif kind == "file":
                console.print(name)
            else:
                console.print(line, style="dim")
        return
    if action == "read_file" and " lines " in result.splitlines()[0]:
        header, _, body = result.partition("\n")
        console.print(f"[dim]{header}[/dim]")
        if not body or body.startswith("(no lines"):
            console.print(body, style="dim")
            return
        code_lines = []
        start = 1
        for index, line in enumerate(body.splitlines()):
            number, sep, code = line.partition("|")
            if sep and number.isdigit():
                if index == 0:
                    start = int(number)
                code_lines.append(code)
            else:
                code_lines.append(line)
        show_code(header.split(" ", 1)[0], "\n".join(code_lines), start)
        return
    if action == "search":
        lines = result.splitlines()
        for line in lines[:40]:
            path, sep, rest = line.partition(":")
            if sep and rest[:1].isdigit():
                number, _, code = rest.partition(":")
                console.print(f"[cyan]{path}[/cyan]:[yellow]{number}[/yellow]: {code}")
            else:
                console.print(line, style="dim")
        if len(lines) > 40:
            console.print(f"[dim]… {len(lines) - 40} more matches[/dim]")
        return
    if result.startswith(("Created ", "Overwrote ", "Edited ")):
        console.print(f"[bold green]{result}[/bold green]")
        return
    if result.startswith(("Refused", "Not ", "No ", "old_str", "Invalid", "Error", "Unknown")):
        preview = result if len(result) < 800 else result[:800] + "…"
        console.print(preview, style="red")
        return
    preview = result if len(result) < 800 else result[:800] + "…"
    console.print(preview, style="dim")


def show_reply(text: str) -> None:
    if not text:
        return
    console.print(Panel(Markdown(text), title="response", title_align="left", border_style="green", padding=(0, 1)))


class Agent:
    def __init__(self, model: str, effort: str, show_reasoning: bool = False):
        self.client = create_client()
        self.model = model
        self.effort = effort
        self.show_reasoning = show_reasoning
        self.messages: list[dict] = [{"role": "system", "content": build_system_prompt()}]

    def complete(self, messages: list[dict]):
        return self.client.beta.chat.completions.parse(
            model=self.model,
            messages=messages,
            response_format=Action,
            reasoning_effort=self.effort,
            max_completion_tokens=MAX_COMPLETION_TOKENS,
        )

    def ask(self, user_text: str) -> str:
        self.messages[0] = {"role": "system", "content": build_system_prompt()}
        expanded = expand_slash(user_text)
        claude = WORKSPACE / "CLAUDE.md"
        claude_text = claude.read_text(encoding="utf-8") if claude.is_file() else ""
        skills = skills_for_texts([user_text, expanded, claude_text])
        for skill in skills:
            show_skill(skill.name)
        if skills:
            expanded += "\n\n" + render_skills(skills)
        self.messages.append({"role": "user", "content": expanded})
        while True:
            show_thinking()
            response = self.complete(self.messages)
            message = response.choices[0].message
            if message.refusal:
                return message.refusal
            step = message.parsed
            if step is None:
                return message.content or "Model returned no action."
            thought = (step.thought or "").strip()
            if thought:
                show_thought(thought)
            self.messages.append({"role": "assistant", "content": step.model_dump_json()})
            if step.action == "done":
                return step.reply or "Done."
            target = step.skill if step.action == "use_skill" else step.path
            show_tool(step.action, target)
            result = run_action(step)
            show_tool_result(step.action, target, result)
            if step.action == "write_file" and result.startswith(("Created ", "Overwrote ")):
                show_code(step.path, step.content)
            elif step.action == "edit_file" and result.startswith("Edited "):
                console.print("[red]- old[/red]")
                show_code(step.path, step.old_str)
                console.print("[green]+ new[/green]")
                show_code(step.path, step.new_str)
            self.messages.append(
                {
                    "role": "user",
                    "content": (
                        "TOOL_RESULT:\n"
                        + result
                        + "\nContinue with the next action, or action=done and a short reply."
                    ),
                }
            )

def render_3d(
    text: str,
    face: str = "#",
    shadow: str = ".",
    depth: int = 2,
    scale: int = 2,
) -> str:
    """Return *text* as bold bitmap letters with a down-right 3D shadow."""
    glyphs = [FONT.get(character, FONT["?"]) for character in text.upper()]
    face_pixels: set[tuple[int, int]] = set()
    cursor = 0

    for glyph in glyphs:
        for y, row in enumerate(glyph):
            for x, pixel in enumerate(row):
                if pixel == "1":
                    # Expanding in both directions gives the letters a bold weight.
                    for scaled_y in range(scale):
                        for scaled_x in range(scale):
                            face_pixels.add(
                                (
                                    (cursor + x) * scale + scaled_x,
                                    y * scale + scaled_y,
                                )
                            )
        cursor += len(glyph[0]) + 1

    if not face_pixels:
        return ""

    width = (cursor - 1) * scale + depth
    height = 7 * scale + depth
    canvas = [[" " for _ in range(width)] for _ in range(height)]

    # Draw every offset layer, from back to front, to make a solid extrusion.
    for offset in range(depth, 0, -1):
        for x, y in face_pixels:
            canvas[y + offset][x + offset] = shadow

    for x, y in face_pixels:
        canvas[y][x] = face

    return "\n".join("".join(row).rstrip() for row in canvas).rstrip()


def single_character(value: str) -> str:
    """Validate argparse options which must contain one character."""
    if len(value) != 1 or value.isspace():
        raise argparse.ArgumentTypeError("must be exactly one non-space character")
    return value

    parser = argparse.ArgumentParser(

    parser.add_argument(

Why is GPT-5.1 slower by default?¶
gpt-5, gpt-5.1, and gpt-5.2 are reasoning models — before producing a final answer, they perform an internal chain-of-thought to break down the problem, evaluate options, and verify their output. This extra "thinking" step improves accuracy and reliability, especially on complex tasks, but comes at the cost of higher latency and token usage.

The tradeoff is straightforward: lowering reasoning effort reduces latency and token cost, but also reduces the model's accuracy on complex tasks. For simple tasks (classification, extraction, chat) this has little to no impact. For multi-step reasoning or analysis, you may notice a drop in quality.

Tune reasoning_effort to match your task (see Reasoning):

Task Complexity	reasoning_effort	Behavior
Complex analysis, multi-step reasoning	high	Full reasoning — highest accuracy, highest latency
General-purpose tasks	medium (default)	Balanced reasoning
Simple chat, extraction, classification	low / minimal	Reduced reasoning — faster, slightly less depth
Latency-critical (autocomplete, routing)	none	No reasoning — behaves like a traditional model, fastest response times
Tip

With reasoning_effort set to none, gpt-5.1 is often faster than gpt-5-nano while being more capable. This makes it a strong single-model choice across a wide range of use cases.

Function Calling / Tool Use¶
Function calling (also known as tool calling) provides a powerful and flexible way for OpenAI models to interface with external systems and access data outside their training data. This guide shows how you can connect a model to data and actions provided by your application. [OpenAI Function Calling Guide]

Function Calling Flow

First let's define a helper function that help us validate tool invocations to prevent hallucination:


import inspect


def check_args(function, args):
    sig = inspect.signature(function)
    params = sig.parameters

    # Check if there are extra arguments
    for name in args:
        if name not in params:
            return False
    # Check if the required arguments are provided
    for name, param in params.items():
        if param.default is param.empty and name not in args:
            return False

    return True

                    },
                    "date": {
                        "type": "date",
                        "description": "The date to get the weather forecast for, in YYYY-MM-DD format.",
                    },
                    "unit": {
                        "type": "string",
                        "description": "The temperature unit (Defaults to 'celcius'). Available types: 'celcius', 'fahrenheit'",
                    }
                },
                "required": ["location", "date"],
            },
        },

response = client.chat.completions.create(
    model="<model-id>",
    messages=messages,
    tools=tools,
    tool_choice="auto",  # auto is default, but we'll be explicit
)

response_message = response.choices[0].message

# check if the model wanted to make a tool call
if tool_calls:

    # extend message with model's response
    messages.append(response_message)

    for tool_call in tool_calls:
        fn_name = tool_call.function.name
        args = tool_call.function.arguments
        fn = {
            "get_current_weather": get_current_weather,
        }.get(fn_name)

        # 1. Validation
        if fn is None:
            raise ValueError(f"Function {fn_name} invoked by model is not a valid function")
        if not check_args(fn, args):
            raise ValueError(f"Function {fn_name} invoked by model has invalid arguments: {args}")

        # 2. Call the tool
        function_response = fn(**args)

        # 3. Extend the messages with function response
        messages.append(
            {
                "tool_call_id": tool_call.id,
                "role": "tool",
                "name": fn_name,
                "content": function_response,
            }
        )

    # 4. Get final response from model
    final_response = client.chat.completions.create(
        model="<model-id>",
        messages=messages,
    )

def process_response_stream_with_tool_calls(stream):
    tool_calls = []
    delta = None

    for chunk in stream:
        delta = chunk.choices[0].delta if chunk.choices and chunk.choices[0].delta is not None else None

        if delta and delta.content:
            print(delta.content, end="", flush=True)

        elif delta and delta.tool_calls:
            tc_chunk_list = delta.tool_calls
            for tc_chunk in tc_chunk_list:
                if len(tool_calls) <= tc_chunk.index:
                    tool_calls.append({"id": "", "type": "function", "function": {"name": "", "arguments": ""}})
                tc = tool_calls[tc_chunk.index]

                if tc_chunk.id:
                    tc["id"] += tc_chunk.id
                if tc_chunk.function.name:
                    tc["function"]["name"] += tc_chunk.function.name
                if tc_chunk.function.arguments:
                    tc["function"]["arguments"] += tc_chunk.function.arguments
